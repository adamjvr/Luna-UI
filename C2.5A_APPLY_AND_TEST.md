```bash
cd ~/GitHub/Luna-UI
git branch --show-current
git status --short
unzip -o ~/Downloads/Luna-UI-C2.5A-Repository-Overlay.zip -d .
```

```bash

```

```bash

```

Inspect and commit only after native graphical validation:

```bash
git add \
  Sources/LunaUI/LunaStaticTextPresentationSnapshot.swift \
  Tests/LunaUIPhase3CTests/LunaC25APresentationSnapshotTests.swift \
  docs/C2.5A_PRESENTATION_SNAPSHOT.md \
```

```bash
LUNA_C25A_SHA="$(git rev-parse HEAD)"
```
